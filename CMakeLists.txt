cmake_minimum_required(VERSION 3.24)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" MACROPULSE_VERSION LIMIT_COUNT 1)
if(NOT MACROPULSE_VERSION MATCHES "^(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)$")
  message(FATAL_ERROR "VERSION must contain a stable semantic version (X.Y.Z)")
endif()
project(MacroPulse VERSION ${MACROPULSE_VERSION} LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "MacroPulse requires Windows 10/11")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(pulse_core src/model.cpp src/engine.cpp src/preferences.cpp src/library.cpp)
target_include_directories(pulse_core PUBLIC src)
target_compile_definitions(pulse_core PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_link_libraries(pulse_core PUBLIC user32 shell32 ole32)
configure_file(src/app.manifest.in generated/app.manifest @ONLY)
configure_file(src/app.rc.in generated/app.rc @ONLY)
add_executable(MacroPulse WIN32 src/main.cpp "${CMAKE_CURRENT_BINARY_DIR}/generated/app.rc")
target_link_libraries(MacroPulse PRIVATE pulse_core comctl32 comdlg32 dwmapi uxtheme gdi32 gdiplus shell32)
if(MSVC)
  foreach(target pulse_core MacroPulse)
    target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /EHsc)
  endforeach()
  target_link_options(MacroPulse PRIVATE /MANIFEST:NO)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(pulse_tests tests/core_tests.cpp)
  target_link_libraries(pulse_tests PRIVATE pulse_core)
  if(MSVC)
    target_compile_options(pulse_tests PRIVATE /W4 /permissive- /utf-8 /EHsc)
  endif()
  add_test(NAME core COMMAND pulse_tests)
  add_executable(library_tests tests/library_tests.cpp)
  target_link_libraries(library_tests PRIVATE pulse_core)
  if(MSVC)
    target_compile_options(library_tests PRIVATE /W4 /permissive- /utf-8 /EHsc)
  endif()
  add_test(NAME library COMMAND library_tests)
  add_test(NAME ui_smoke COMMAND MacroPulse --smoke-test)
  add_test(NAME library_storage COMMAND MacroPulse --library-storage-test "${CMAKE_CURRENT_BINARY_DIR}/test-library/settings.dat")
  add_test(NAME session_write COMMAND MacroPulse --session-write-test "${CMAKE_CURRENT_BINARY_DIR}/test-session/settings.dat")
  add_test(NAME session_read COMMAND MacroPulse --session-read-test "${CMAKE_CURRENT_BINARY_DIR}/test-session/settings.dat")
  set_tests_properties(session_write PROPERTIES FIXTURES_SETUP saved_session)
  set_tests_properties(session_read PROPERTIES FIXTURES_REQUIRED saved_session)
  set_tests_properties(core library ui_smoke library_storage session_write session_read PROPERTIES TIMEOUT 120)
endif()
