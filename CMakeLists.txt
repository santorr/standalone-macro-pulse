cmake_minimum_required(VERSION 3.24)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" MACROPULSE_VERSION LIMIT_COUNT 1)
if(NOT MACROPULSE_VERSION MATCHES "^(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)$")
  message(FATAL_ERROR "VERSION must contain a stable semantic version (X.Y.Z)")
endif()
project(MacroPulse VERSION ${MACROPULSE_VERSION} LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "MacroPulse requires Windows 10/11")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(pulse_core src/model.cpp src/engine.cpp src/preferences.cpp src/library.cpp src/text_focus.cpp src/mouse_shortcuts.cpp)
target_include_directories(pulse_core PUBLIC src)
target_compile_definitions(pulse_core PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_link_libraries(pulse_core PUBLIC user32 shell32 ole32 oleaut32 uiautomationcore)
configure_file(src/version.hpp.in generated/version.hpp @ONLY)
add_library(pulse_updater src/updater.cpp)
target_include_directories(pulse_updater PUBLIC src "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_definitions(pulse_updater PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_link_libraries(pulse_updater PUBLIC winhttp bcrypt version windowsapp ole32 user32)
configure_file(src/app.manifest.in generated/app.manifest @ONLY)
configure_file(src/app.rc.in generated/app.rc @ONLY)
add_executable(MacroPulse WIN32 src/main.cpp "${CMAKE_CURRENT_BINARY_DIR}/generated/app.rc")
target_link_libraries(MacroPulse PRIVATE pulse_core pulse_updater comctl32 comdlg32 dwmapi uxtheme gdi32 gdiplus shell32)
if(MSVC)
  foreach(target pulse_core pulse_updater MacroPulse)
    target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /EHsc)
  endforeach()
  target_link_options(MacroPulse PRIVATE /MANIFEST:NO)
endif()
include(CTest)
if(BUILD_TESTING)
  math(EXPR TEST_UPGRADE_MAJOR "${PROJECT_VERSION_MAJOR} + 1")
  configure_file(tests/update_test_payload.rc.in generated/update_test_payload.rc @ONLY)
  configure_file(tests/update_test_version.hpp.in generated/update_test_version.hpp @ONLY)
  add_executable(update_test_payload WIN32 tests/update_test_payload.cpp "${CMAKE_CURRENT_BINARY_DIR}/generated/update_test_payload.rc")
  target_compile_definitions(update_test_payload PRIVATE UNICODE _UNICODE)
  add_executable(update_test_host tests/update_test_host.cpp)
  target_link_libraries(update_test_host PRIVATE pulse_updater)
  add_executable(updater_tests tests/updater_tests.cpp)
  target_link_libraries(updater_tests PRIVATE pulse_updater)
  add_dependencies(updater_tests MacroPulse update_test_host update_test_payload)
  if(MSVC)
    foreach(target updater_tests update_test_host update_test_payload)
      target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /EHsc)
    endforeach()
  endif()
  add_test(NAME updater COMMAND updater_tests "$<TARGET_FILE:MacroPulse>" "$<TARGET_FILE:update_test_payload>" "${CMAKE_CURRENT_BINARY_DIR}/test-updater")
  add_executable(mouse_shortcut_tests tests/mouse_shortcut_tests.cpp)
  target_link_libraries(mouse_shortcut_tests PRIVATE pulse_core)
  if(MSVC)
    target_compile_options(mouse_shortcut_tests PRIVATE /W4 /permissive- /utf-8 /EHsc)
  endif()
  add_test(NAME mouse_shortcuts COMMAND mouse_shortcut_tests)
  add_executable(text_focus_tests tests/text_focus_tests.cpp)
  target_link_libraries(text_focus_tests PRIVATE pulse_core)
  if(MSVC)
    target_compile_options(text_focus_tests PRIVATE /W4 /permissive- /utf-8 /EHsc)
  endif()
  add_test(NAME text_focus COMMAND text_focus_tests)
  add_executable(pulse_tests tests/core_tests.cpp)
  target_link_libraries(pulse_tests PRIVATE pulse_core)
  if(MSVC)
    target_compile_options(pulse_tests PRIVATE /W4 /permissive- /utf-8 /EHsc)
  endif()
  add_test(NAME core COMMAND pulse_tests)
  add_executable(library_tests tests/library_tests.cpp)
  target_link_libraries(library_tests PRIVATE pulse_core)
  if(MSVC)
    target_compile_options(library_tests PRIVATE /W4 /permissive- /utf-8 /EHsc)
  endif()
  add_test(NAME library COMMAND library_tests)
  add_test(NAME render_regression COMMAND MacroPulse --render-regression)
  add_test(NAME ui_smoke COMMAND MacroPulse --smoke-test)
  add_test(NAME library_storage COMMAND MacroPulse --library-storage-test "${CMAKE_CURRENT_BINARY_DIR}/test-library/settings.dat")
  add_test(NAME session_write COMMAND MacroPulse --session-write-test "${CMAKE_CURRENT_BINARY_DIR}/test-session/settings.dat")
  add_test(NAME session_read COMMAND MacroPulse --session-read-test "${CMAKE_CURRENT_BINARY_DIR}/test-session/settings.dat")
  set_tests_properties(session_write PROPERTIES FIXTURES_SETUP saved_session)
  set_tests_properties(session_read PROPERTIES FIXTURES_REQUIRED saved_session)
  set_tests_properties(render_regression mouse_shortcuts text_focus updater core library ui_smoke library_storage session_write session_read PROPERTIES TIMEOUT 120)
endif()
